def get_stock_data(ticker: str) -> pd.DataFrame:
    """Download historical stock prices (Date, Open, High, Low, Close, Volume) with yfinance."""
    df = yf.download(ticker, progress=False)
    df = df.reset_index()
    df['Date'] = pd.to_datetime(df['Date'])
    # Keep only the columns used by the assignment
    return df[['Date', 'Open', 'High', 'Low', 'Close', 'Volume']]

def scrape_quarterly_revenue(page_url: str, company_keyword: str) -> pd.DataFrame:
    """
    Scrape quarterly revenue table for a given company from the IBM S3 revenue page.
    We search for a heading containing the company name and the words 'Quarterly Revenue',
    then pick the next <table> element.
    Returns a DataFrame with columns ['Date', 'Revenue'] (Revenue as numeric).
    """
    html = requests.get(page_url).text
    soup = BeautifulSoup(html, 'html.parser')

    # Locate the heading that matches the company keyword + 'Quarterly Revenue'
    heading = None
    for tag in soup.find_all(['h1','h2','h3','h4','h5','h6']):
        text = tag.get_text(strip=True)
        if company_keyword.lower() in text.lower() and 'quarterly revenue' in text.lower():
            heading = tag
            break
    if heading is None:
        # Fallback: try to find all tables and choose the one with 2 columns & header including 'Revenue'
        for tbl in soup.find_all('table'):
            ths = [th.get_text(strip=True).lower() for th in tbl.find_all('th')]
            if len(ths) >= 2 and 'revenue' in ''.join(ths).lower():
                target_table = tbl
                break
        else:
            raise ValueError("Quarterly revenue table not found.")
    else:
        # Next table sibling after the heading
        nxt = heading.find_next('table')
        if nxt is None:
            raise ValueError("Table following the heading not found.")
        target_table = nxt

    # Parse rows
    rows = []
    for tr in target_table.tbody.find_all('tr'):
        tds = tr.find_all('td')
        if len(tds) >= 2:
            date = tds[0].get_text(strip=True)
            revenue = tds[1].get_text(strip=True)
            if revenue:
                rows.append([date, revenue])

    df = pd.DataFrame(rows, columns=['Date', 'Revenue'])

    # Clean Revenue: remove $ and , then convert to numeric (drop blanks)
    df['Revenue'] = (df['Revenue'].astype(str)
                     .str.replace(r'[\$,]', '', regex=True)
                     .replace('', pd.NA))
    df = df.dropna(subset=['Revenue'])
    df['Revenue'] = pd.to_numeric(df['Revenue'], errors='coerce')
    df = df.dropna(subset=['Revenue']).reset_index(drop=True)
    # Parse dates
    df['Date'] = pd.to_datetime(df['Date'])
    return df

def make_graph(stock_data: pd.DataFrame, revenue_data: pd.DataFrame, stock_name: str):
    """Plot Close price and Quarterly Revenue up to June 2021."""
    s = stock_data.copy()
    r = revenue_data.copy()

    s['Date'] = pd.to_datetime(s['Date'])
    r['Date'] = pd.to_datetime(r['Date'])
    r['Revenue'] = pd.to_numeric(r['Revenue'], errors='coerce')

    cutoff = pd.Timestamp('2021-06-30')
    s = s[s['Date'] <= cutoff]
    r = r[r['Date'] <= cutoff]

    # Stock price
    plt.figure(figsize=(10,5))
    plt.plot(s['Date'], s['Close'])
    plt.title(f'{stock_name} Stock Price (Close)')
    plt.xlabel('Date'); plt.ylabel('Price'); plt.grid(True)
    plt.show()

    # Quarterly revenue
    plt.figure(figsize=(10,5))
    plt.plot(r['Date'], r['Revenue'])
    plt.title(f'{stock_name} Quarterly Revenue')
    plt.xlabel('Date'); plt.ylabel('Revenue'); plt.grid(True)
    plt.show()
